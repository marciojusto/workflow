```typescript
import mongoose, { Model, Schema } from 'mongoose'

export interface IVatSettings {
  country: string
  vatCode: string
  enumId: string
  percentage: number
  validFrom: Date
}

const VatSettingsSchema = new Schema<IVatSettings>({
  country: { type: String, required: true },
  vatCode: { type: String, required: true },
  enumId: { type: String, required: true },
  percentage: { type: Number, required: true },
  validFrom: { type: Date, default: Date.now }
})

// Unique index: country + enumId
VatSettingsSchema.index({ country: 1, enumId: 1 }, { unique: true })

export const VatSettings: Model<IVatSettings> = 
  mongoose.models.VatSettings || mongoose.model('VatSettings', VatSettingsSchema)
```

```javascript
// server/scripts/insertVatData.cjs
import fs from 'fs'
import path from 'path'
import mongoose from 'mongoose'

// CSV parsing (semicolon-separated)
function parseCSV(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8')
  const lines = content.trim().split('\n')
  const headers = lines[0].split(';').map(h => h.replace(/"/g, '').trim())
  
  return lines.slice(1).map(line => {
    const values = line.split(';').map(v => v.replace(/"/g, '').trim())
    const obj = {}
    headers.forEach((header, i) => {
      obj[header] = values[i]
    })
    return obj
  })
}

async function main() {
  // Connect to MongoDB
  const MONGODB_URI = process.env.MONGODB_URI || 'mongodb://localhost:27017/hyperfront'
  await mongoose.connect(MONGODB_URI)
  console.log('Connected to MongoDB')

  // Read CSV
  const csvPath = path.join(__dirname, 'data', 'vat.csv')
  const records = parseCSV(csvPath)
  console.log(`Parsed ${records.length} records from CSV`)

  // Insert/Update records
  const VatSettings = mongoose.model('VatSettings', new mongoose.Schema({
    country: String,
    vatCode: String,
    enumId: String,
    percentage: Number,
    validFrom: Date
  }, { collection: 'vat_settings' }))

  let inserted = 0
  let updated = 0

  for (const record of records) {
    // Skip rows without enumId or percentage
    if (!record.ID || !record['VAT Tariff']) continue

    const percentage = parseFloat(record['VAT Tariff'].replace(',', '.'))
    if (isNaN(percentage)) continue

    const doc = {
      country: record['Country.Country Name'],
      vatCode: record['VAT Code'],
      enumId: record.ID,
      percentage: percentage,
      validFrom: new Date(record.validFrom)
    }

    // Upsert
    const result = await VatSettings.findOneAndUpdate(
      { country: doc.country, enumId: doc.enumId },
      doc,
      { upsert: true, new: true }
    )

    if (result.isNew) {
      inserted++
    } else {
      updated++
    }
  }

  console.log(`Done: ${inserted} inserted, ${updated} updated`)
  await mongoose.disconnect()
}

main().catch(console.error)
```

```typescript
async function getVatPercentageFromMongo(enumId: string | undefined): Promise<number | null> {
  if (!enumId) return null
  
  try {
    const vatSetting = await VatSettings.findOne({ enumId })
    return vatSetting?.percentage ?? null
  } catch (err) {
    console.error('Failed to lookup VAT percentage:', err)
    return null
  }
}
```

```typescript
// Adicionar computed property
const vatPercentageDisplay = computed(() => {
  const pd = pricingDetails.value
  const pct = pd?.vatPercentage ?? pd?.vat ?? null
  if (pct != null) {
    // Handle decimal (0.22) or percentage (22)
    const num = sanitizeToNumber(pct)
    if (num > 0 && num <= 1) return `${(num * 100).toFixed(0)}%`
    if (num >= 0 && num <= 100) return `${num.toFixed(0)}%`
  }
  return '-'
})
```
